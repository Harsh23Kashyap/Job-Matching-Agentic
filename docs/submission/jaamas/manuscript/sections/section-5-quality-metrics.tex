\subsection{Similarity and component scores}\label{sec:5-scoring}

All component scores are normalized to $[0,1]$.
Let $\mathbf{e}_c, \mathbf{e}_j \in \mathbb{R}^{d}$ denote snapshot embeddings ($d{=}384$).

\leadpara{Semantic similarity.}
Cosine score:
\begin{equation}
  s_{\mathrm{sem}}^{\mathrm{cos}} =
    \frac{\mathbf{e}_c \cdot \mathbf{e}_j}{\|\mathbf{e}_c\|\,\|\mathbf{e}_j\|}.
  \label{eq:semantic-cosine}
\end{equation}
Euclidean-derived alternative: $s_{\mathrm{sem}}^{\mathrm{euc}} = 1 / (1 + \|\mathbf{e}_c - \mathbf{e}_j\|_2)$.

\leadpara{Skill overlap.}
For canonical skill sets $A$ (resume) and $B$ (required job skills), Jaccard mode:
\begin{equation}
  s_{\mathrm{skl}}^{\mathrm{jac}} =
    |A \cap B| / |A \cup B| \quad (\text{0 if either set empty}).
  \label{eq:jaccard}
\end{equation}
Soft-embed mode averages, over each $b \in B$, the maximum cosine between skill phrase embeddings.

\leadpara{Composite score (portal default).}
\begin{equation}
  s_{\mathrm{final}} =
    \sum_{k} w_k s_k, \quad
    (w_{\mathrm{sem}}, w_{\mathrm{skl}}, w_{\mathrm{tit}}, w_{\mathrm{exp}}, w_{\mathrm{comp}}, w_{\mathrm{rem}})
    = (0.28, 0.27, 0.10, 0.15, 0.10, 0.10),
  \label{eq:composite}
\end{equation}
with $s_k$ the semantic, skill, title, experience, compensation, and remote components; result clipped to $[0,1]$.

\leadpara{Multimodal baseline.}
$s_{\mathrm{mm}} = \alpha s_{\mathrm{sem}} + (1-\alpha) s_{\mathrm{skl}}$ with default $\alpha{=}0.7$.

\leadpara{Reciprocal rank fusion.}
For ranked lists $\mathcal{L}_1,\ldots,\mathcal{L}_m$:
\begin{equation}
  \mathrm{RRF}(x) = \sum_{i=1}^{m} \sum_{r} \mathbb{1}[\mathcal{L}_i[r] = x] \cdot \frac{1}{k + r},
  \label{eq:rrf}
\end{equation}
with $k{=}60$ by default.

\leadpara{Optional adjustments.}
Hard constraints multiply $s_{\mathrm{final}}$ by feasibility factor $\phi \in (0,1]$ when enabled.
Platt calibration: $\hat{p}(s) = \sigma(a s + b)$ with fitted $(a,b)$.
Feedback boost adds weighted save/apply/dismiss deltas when enabled (off in default portal).

\subsection{Ranking metrics}\label{sec:5-ranking-metrics}

Given ranked list $\pi_q$ for query $q$ and relevant set $\mathcal{R}_q = \{ j : \mathrm{rel}_q(j) > 0 \}$, with top-$K$ prefix $\pi_q[:K]$ and default $K{=}5$:

\begin{align}
  \mathrm{P@}K(q) &= \frac{1}{K}\sum_{j \in \pi_q[:K]} \mathbb{1}[j \in \mathcal{R}_q],
  \label{eq:p-at-k}\\
  \mathrm{R@}K(q) &= \frac{1}{|\mathcal{R}_q|}\sum_{j \in \pi_q[:K]} \mathbb{1}[j \in \mathcal{R}_q],
  \label{eq:r-at-k}
\end{align}

Graded nDCG:
\begin{equation}
  \mathrm{nDCG@}K(q) = \frac{\sum_{i=1}^{K} (2^{\mathrm{rel}_q(j_i)} - 1) / \log_2(i+1)}
  {\mathrm{DCG@}K(q^\ast)},
  \label{eq:ndcg}
\end{equation}
where $\pi_{q^\ast}$ is the ideal ranking.
MRR and MAP treat relevance as binary ($r>0$).
Metrics are macro-averaged over queries; significance tests use 5000 bootstrap resamples (seed~42).

\leadpara{Hard-negative mining.}
From top-$P{=}10$ multimodal pool, up to five unlabeled high-scoring jobs per query:
\begin{equation}
  \mathcal{H}_q = \left\{ j \in \pi_q[:P] : j \notin \mathcal{R}_q \right\}_{|\cdot|\leq 5}.
  \label{eq:hard-negs}
\end{equation}
